# ==============================================================================
# KodeKloud Azure Playground - AKS + OpenText VTS 26.1 Repro Commands
# ==============================================================================
# Notes:
# - Tested on KodeKloud Azure Playground (Standard_D2s_v3, managed-csi, Kubernetes 1.36)
# - Premium disks are blocked in KodeKloud; use managed-csi (Standard SSD, <= 128 GB)
# - VTS Basic Auth returns 401 on /, so probes use tcpSocket on port 8443
# ==============================================================================

# 1. Verify pre-assigned Resource Group & Location (do NOT create a new RG)
az group list -o table
RG=$(az group list --query "[0].name" -o tsv)
LOC=$(az group list --query "[0].location" -o tsv)
echo "Using RG=$RG in Location=$LOC"

# 2. Provision 2-node AKS cluster (~4-5 mins)
az aks create \
  --resource-group "$RG" \
  --name vts-aks-test \
  --location "$LOC" \
  --node-count 2 \
  --node-vm-size Standard_D2s_v3 \
  --generate-ssh-keys

# 3. Connect kubectl & verify nodes + storageclasses
az aks get-credentials --resource-group "$RG" --name vts-aks-test --overwrite-existing
kubectl get nodes
kubectl get sc

# 4. Deploy OpenText VTS 26.1 (Hardened non-root, managed-csi PVC, tcpSocket probes, LoadBalancer)
cat <<'EOF' | kubectl apply -f -
apiVersion: v1
kind: Namespace
metadata:
  name: lre-vts
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: vts-app1-data-pvc
  namespace: lre-vts
spec:
  accessModes:
    - ReadWriteOnce
  storageClassName: managed-csi
  resources:
    requests:
      storage: 10Gi
---
apiVersion: v1
kind: ConfigMap
metadata:
  name: vts-app1-config
  namespace: lre-vts
data:
  configure.json: |
    {
      "dbPath": "/var/lib/OT/OT_VTS/db/data",
      "defaultDbName": "Default",
      "adminPort": 4000,
      "defaultApiPort": 8888,
      "defaultStart": true,
      "enableDiag": false,
      "autoRefreshGrid": false,
      "maxInstancesAllowed": 50,
      "autoCreateIndexedColumn": true,
      "normalizeCSV": false,
      "admin": {
        "authentication": "Basic",
        "users": [
          { "user": "vtsadmin", "password": "SandboxTest123!", "display": "VTS Administrator" }
        ],
        "role": { "administrator": ["vtsadmin"], "user": [] },
        "requireBasicAuthForAPI": true,
        "useSSL": false
      },
      "defaultLanguage": "en",
      "dbInfo": {
        "connPool": { "maxIdle": 5000, "maxActive": 5000 },
        "dbServerPort": { "start": 10000, "end": 30000 },
        "dbCheck": { "waitingTime": 10, "retryTimes": 3600 }
      },
      "logger": {
        "level": "error",
        "transports": { "console": { "type": "Console" } }
      }
    }
  vts-nginx.conf: |
    worker_processes auto;
    error_log /tmp/nginx_error.log warn;
    pid /tmp/nginx.pid;
    events { worker_connections 4096; }
    http {
        include /etc/nginx/mime.types;
        default_type application/octet-stream;
        sendfile on;
        keepalive_timeout 65;
        map $http_upgrade $connection_upgrade { default upgrade; '' ''; }
        server {
            listen 8443 ssl;
            server_name _;
            ssl_certificate /opt/OT/OT_VTS/web/vts.crt;
            ssl_certificate_key /opt/OT/OT_VTS/web/vts.key;
            ssl_protocols TLSv1.2 TLSv1.3;
            ssl_ciphers HIGH:!aNULL:!MD5;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection $connection_upgrade;
            proxy_set_header Host $host;
            proxy_set_header X-Forwarded-Proto https;
            location / {
                if ($query_string ~* "^portnumber=(4000|888[89]|889[0-9]|890[0-8]|20[0-4][0-9]|2050)(.*)$") {
                    set $inst_port $1;
                    set $inst_param $2;
                    proxy_pass http://127.0.0.1:$inst_port$inst_param;
                    break;
                }
                if ($query_string ~* "portnumber=") { return 403; }
                proxy_pass http://127.0.0.1:4000;
            }
        }
    }
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: vts-app1-team
  namespace: lre-vts
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app: vts-app1-team
  template:
    metadata:
      labels:
        app: vts-app1-team
    spec:
      securityContext:
        runAsNonRoot: true
        runAsUser: 54345
        runAsGroup: 54345
        fsGroup: 54345
        seccompProfile:
          type: RuntimeDefault
      containers:
      - name: vts-server
        image: performancetesting/opentext_vts_linux_ubuntu:26.1
        imagePullPolicy: IfNotPresent
        workingDir: /opt/OT/OT_VTS/web
        securityContext:
          allowPrivilegeEscalation: false
          runAsNonRoot: true
          runAsUser: 54345
          runAsGroup: 54345
          capabilities:
            drop: ["ALL"]
        env:
        - name: GIN_MODE
          value: "release"
        resources:
          requests:
            cpu: "500m"
            memory: "1Gi"
          limits:
            cpu: "1500m"
            memory: "3Gi"
        command: ["/bin/bash", "-c"]
        args:
        - |
          export GIN_MODE=release
          if [ ! -f /opt/OT/OT_VTS/web/vts.crt ] || [ ! -f /opt/OT/OT_VTS/web/vts.key ]; then
            openssl req -x509 -nodes -days 3650 -newkey rsa:2048 \
              -keyout /opt/OT/OT_VTS/web/vts.key \
              -out /opt/OT/OT_VTS/web/vts.crt \
              -subj "/CN=vts-app1.aks.test" 2>/dev/null || true
          fi
          DATA_FILE="/var/lib/OT/OT_VTS/db/data/instances.json"
          mkdir -p /var/lib/OT/OT_VTS/db/data
          [ -f "$DATA_FILE" ] || echo "[]" > "$DATA_FILE"
          ln -sf "$DATA_FILE" /opt/OT/OT_VTS/web/instances.json
          sed -i '/^save /d' /opt/OT/OT_VTS/web/redis.conf 2>/dev/null || true
          echo "save 60 1" >> /opt/OT/OT_VTS/web/redis.conf 2>/dev/null || true
          nginx
          ./api_daemon </dev/null &
          wait -n
        ports:
        - name: https-web
          containerPort: 8443
        startupProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 5
          failureThreshold: 24
        readinessProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 10
          timeoutSeconds: 3
          failureThreshold: 3
        livenessProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 20
          timeoutSeconds: 3
          failureThreshold: 3
        volumeMounts:
        - name: config-mount
          mountPath: /opt/OT/OT_VTS/web/configure.json
          subPath: configure.json
        - name: nginx-mount
          mountPath: /etc/nginx/nginx.conf
          subPath: vts-nginx.conf
        - name: vts-db-storage
          mountPath: /var/lib/OT/OT_VTS/db/data
      volumes:
      - name: config-mount
        configMap:
          name: vts-app1-config
          items:
          - key: configure.json
            path: configure.json
      - name: nginx-mount
        configMap:
          name: vts-app1-config
          items:
          - key: vts-nginx.conf
            path: vts-nginx.conf
      - name: vts-db-storage
        persistentVolumeClaim:
          claimName: vts-app1-data-pvc
---
apiVersion: v1
kind: Service
metadata:
  name: vts-app1-team-service
  namespace: lre-vts
spec:
  type: LoadBalancer
  ports:
  - name: https-web
    port: 443
    targetPort: 8443
    protocol: TCP
  selector:
    app: vts-app1-team
EOF

# 5. Check pod, PVC, and LoadBalancer External IP (open https://<EXTERNAL-IP> in browser)
kubectl get pods,pvc,svc -n lre-vts

# 6. Persistence test after creating a table/instance in UI (vtsadmin / SandboxTest123!)
kubectl delete pod -n lre-vts -l app=vts-app1-team
kubectl get pods -n lre-vts -w
