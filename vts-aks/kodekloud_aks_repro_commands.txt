# ==============================================================================
# KodeKloud Azure Playground - AKS + OpenText VTS 26.1 Repro Commands
# ==============================================================================
# Matches the current vts-aks/ architecture:
# - Separate Secret (vts-app1-credentials) with administrator & read-only user roles
# - Separate TLS Secret (vts-app1-tls-cert) mounted read-only at /opt/OT/OT_VTS/web/certs
# - ConfigMap template rendered at startup into /opt/OT/OT_VTS/web/configure.json (mode 0600)
# - CVE-remediated NGINX (listen 8443 ssl; AES-256-GCM / AES-128-GCM ciphers)
# - Hardened Pod Security (runAsNonRoot: true, UID/GID 54345, drop ALL, RuntimeDefault)
# - tcpSocket probes on port 8443 + managed-csi (Standard SSD) PVC + preStop Valkey flush
# ==============================================================================

# 1. Verify pre-assigned Resource Group & Location (do NOT create a new RG)
az group list -o table
RG=$(az group list --query "[0].name" -o tsv)
LOC=$(az group list --query "[0].location" -o tsv)
echo "Using RG=$RG in Location=$LOC"

# 2. Provision 2-node AKS cluster (~4-5 mins)
az aks create \
  --resource-group "$RG" \
  --name vts-aks-test \
  --location "$LOC" \
  --node-count 2 \
  --node-vm-size Standard_D2s_v3 \
  --generate-ssh-keys

# 3. Connect kubectl & verify nodes + storageclasses
az aks get-credentials --resource-group "$RG" --name vts-aks-test --overwrite-existing
kubectl get nodes
kubectl get sc

# 4. Create Namespace & Sandbox TLS Secret (simulates Enterprise PKI Secret mount)
kubectl create namespace lre-vts
openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout /tmp/vts-tls.key -out /tmp/vts-tls.crt \
  -subj "/CN=vts-app1.internal.cloud.corp"
kubectl create secret tls vts-app1-tls-cert -n lre-vts \
  --cert=/tmp/vts-tls.crt --key=/tmp/vts-tls.key
rm -f /tmp/vts-tls.crt /tmp/vts-tls.key

# 5. Deploy PVC, Dual-Role Secret, ConfigMap Template, Hardened Deployment & LoadBalancer
cat <<'EOF' | kubectl apply -f -
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: vts-app1-data-pvc
  namespace: lre-vts
spec:
  accessModes:
    - ReadWriteOnce
  storageClassName: managed-csi
  resources:
    requests:
      storage: 10Gi
---
apiVersion: v1
kind: Secret
metadata:
  name: vts-app1-credentials
  namespace: lre-vts
type: Opaque
stringData:
  admin_user: "vtsadmin"
  admin_password: "SandboxAdmin123!"
  readonly_user: "tester"
  readonly_password: "SandboxUser123!"
---
apiVersion: v1
kind: ConfigMap
metadata:
  name: vts-app1-config
  namespace: lre-vts
data:
  configure.json: |
    {
      "dbPath": "/var/lib/OT/OT_VTS/db/data",
      "defaultDbName": "Default",
      "adminPort": 4000,
      "defaultApiPort": 8888,
      "defaultStart": true,
      "enableDiag": false,
      "autoRefreshGrid": false,
      "maxInstancesAllowed": 50,
      "autoCreateIndexedColumn": true,
      "normalizeCSV": false,
      "admin": {
        "authentication": "Basic",
        "users": [
          { "user": "vtsadmin", "password": "__VTS_ADMIN_PASSWORD__", "display": "VTS Administrator" },
          { "user": "tester", "password": "__VTS_READONLY_PASSWORD__", "display": "Test Operator (Read-Only)" }
        ],
        "role": { "administrator": ["vtsadmin"], "user": ["tester"] },
        "requireBasicAuthForAPI": true,
        "useSSL": false
      },
      "defaultLanguage": "en",
      "dbInfo": {
        "connPool": { "maxIdle": 5000, "maxActive": 5000 },
        "dbServerPort": { "start": 10000, "end": 30000 },
        "dbCheck": { "waitingTime": 10, "retryTimes": 3600 }
      },
      "logger": {
        "level": "error",
        "transports": { "console": { "type": "Console" } }
      }
    }
  vts-nginx.conf: |
    worker_processes auto;
    error_log /tmp/nginx_error.log warn;
    pid /tmp/nginx.pid;
    events { worker_connections 4096; }
    http {
        include /etc/nginx/mime.types;
        default_type application/octet-stream;
        sendfile on;
        keepalive_timeout 65;
        map $http_upgrade $connection_upgrade { default upgrade; '' ''; }
        server {
            listen 8443 ssl;
            server_name _;
            ssl_certificate /opt/OT/OT_VTS/web/certs/tls.crt;
            ssl_certificate_key /opt/OT/OT_VTS/web/certs/tls.key;
            ssl_protocols TLSv1.2 TLSv1.3;
            ssl_ciphers ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection $connection_upgrade;
            proxy_set_header Host $host;
            proxy_set_header X-Forwarded-Proto https;
            location / {
                if ($query_string ~* "^portnumber=(4000|888[89]|889[0-9]|890[0-8]|20[0-4][0-9]|2050)(.*)$") {
                    set $inst_port $1;
                    set $inst_param $2;
                    proxy_pass http://127.0.0.1:$inst_port$inst_param;
                    break;
                }
                if ($query_string ~* "portnumber=") { return 403; }
                proxy_pass http://127.0.0.1:4000;
            }
        }
    }
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: vts-app1-team
  namespace: lre-vts
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app: vts-app1-team
  template:
    metadata:
      labels:
        app: vts-app1-team
    spec:
      securityContext:
        runAsNonRoot: true
        runAsUser: 54345
        runAsGroup: 54345
        fsGroup: 54345
        seccompProfile:
          type: RuntimeDefault
      containers:
      - name: vts-server
        image: performancetesting/opentext_vts_linux_ubuntu:26.1
        imagePullPolicy: IfNotPresent
        workingDir: /opt/OT/OT_VTS/web
        securityContext:
          allowPrivilegeEscalation: false
          runAsNonRoot: true
          runAsUser: 54345
          runAsGroup: 54345
          capabilities:
            drop: ["ALL"]
        env:
        - name: GIN_MODE
          value: "release"
        - name: VTS_ADMIN_PASSWORD
          valueFrom:
            secretKeyRef:
              name: vts-app1-credentials
              key: admin_password
        - name: VTS_READONLY_PASSWORD
          valueFrom:
            secretKeyRef:
              name: vts-app1-credentials
              key: readonly_password
        resources:
          requests:
            cpu: "500m"
            memory: "1Gi"
          limits:
            cpu: "1500m"
            memory: "3Gi"
        command: ["/bin/bash", "-c"]
        args:
        - |
          export GIN_MODE=release
          /opt/OT/OT_VTS/web/node -e '
          const fs = require("fs");
          const adminPwd = process.env.VTS_ADMIN_PASSWORD || "";
          const readonlyPwd = process.env.VTS_READONLY_PASSWORD || "";
          if (!adminPwd || !readonlyPwd) { console.error("Missing VTS Secret passwords"); process.exit(1); }
          const rendered = fs.readFileSync("/tmp/configure.json.tmpl", "utf8")
            .replace("__VTS_ADMIN_PASSWORD__", adminPwd)
            .replace("__VTS_READONLY_PASSWORD__", readonlyPwd);
          fs.writeFileSync("/opt/OT/OT_VTS/web/configure.json", rendered, { mode: 0o600 });
          '
          DATA_FILE="/var/lib/OT/OT_VTS/db/data/instances.json"
          mkdir -p /var/lib/OT/OT_VTS/db/data
          [ -f "$DATA_FILE" ] || echo "[]" > "$DATA_FILE"
          ln -sf "$DATA_FILE" /opt/OT/OT_VTS/web/instances.json
          sed -i '/^save /d' /opt/OT/OT_VTS/web/redis.conf 2>/dev/null || true
          echo "save 60 1" >> /opt/OT/OT_VTS/web/redis.conf 2>/dev/null || true
          nginx
          ./api_daemon </dev/null &
          wait -n
        lifecycle:
          preStop:
            exec:
              command:
              - /bin/bash
              - -c
              - |
                if [ -s /var/lib/OT/OT_VTS/db/data/instances.json ]; then
                  cp -f /var/lib/OT/OT_VTS/db/data/instances.json /var/lib/OT/OT_VTS/db/data/instances.json.bak
                fi
                for pid in $(pgrep valkey-server 2>/dev/null); do
                  PORT=$(cat /proc/$pid/cmdline 2>/dev/null | tr '\0' ' ' | grep -oP '127\.0\.0\.1:\K\d+')
                  if [ -n "$PORT" ]; then
                    /opt/OT/OT_VTS/web/valkey-cli -p "$PORT" save 2>/dev/null || true
                  fi
                done
                sync
        ports:
        - name: https-web
          containerPort: 8443
        startupProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 5
          failureThreshold: 24
        readinessProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 10
          timeoutSeconds: 3
          failureThreshold: 3
        livenessProbe:
          tcpSocket:
            port: 8443
          periodSeconds: 20
          timeoutSeconds: 3
          failureThreshold: 3
        volumeMounts:
        - name: config-mount
          mountPath: /tmp/configure.json.tmpl
          subPath: configure.json
        - name: nginx-mount
          mountPath: /etc/nginx/nginx.conf
          subPath: vts-nginx.conf
        - name: vts-db-storage
          mountPath: /var/lib/OT/OT_VTS/db/data
        - name: enterprise-tls
          mountPath: /opt/OT/OT_VTS/web/certs
          readOnly: true
      volumes:
      - name: config-mount
        configMap:
          name: vts-app1-config
          items:
          - key: configure.json
            path: configure.json
      - name: nginx-mount
        configMap:
          name: vts-app1-config
          items:
          - key: vts-nginx.conf
            path: vts-nginx.conf
      - name: vts-db-storage
        persistentVolumeClaim:
          claimName: vts-app1-data-pvc
      - name: enterprise-tls
        secret:
          secretName: vts-app1-tls-cert
---
apiVersion: v1
kind: Service
metadata:
  name: vts-app1-team-service
  namespace: lre-vts
spec:
  type: LoadBalancer
  ports:
  - name: https-web
    port: 443
    targetPort: 8443
    protocol: TCP
  selector:
    app: vts-app1-team
EOF
